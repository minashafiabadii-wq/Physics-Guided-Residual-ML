from xgboost import XGBRegressor

def build_model(random_state=42):
    return XGBRegressor(n_estimators=25, max_depth=4, learning_rate=0.08, subsample=0.85, colsample_bytree=0.85, reg_lambda=1.0, objective='reg:squarederror', tree_method='hist', max_bin=128, random_state=random_state, n_jobs=4, verbosity=0)
